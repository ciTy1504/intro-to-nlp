\section{Các phương pháp tinh chỉnh (Fine-tuning)}
\label{sec:finetuning_methods}

\subsection{Học chuyển giao (Transfer Learning)}
\label{ssec:transfer_learning}
Nền tảng của mọi kỹ thuật tinh chỉnh là \textbf{học chuyển giao (transfer learning)}.

\begin{definition}{Học chuyển giao}{def:transfer_learning}
    Học chuyển giao là một phương pháp học máy trong đó một mô hình được phát triển cho một tác vụ A được tái sử dụng làm điểm khởi đầu cho một mô hình trong tác vụ B.
\end{definition}

Trong NLP, học chuyển giao là mô hình (paradigm) thống trị, gồm hai giai đoạn:
\begin{enumerate}
    \item \textbf{Huấn luyện trước (pre-training).} Huấn luyện một mô hình lớn (như BERT, GPT) trên một tác vụ chung (ví dụ: Masked Language Modeling, Next Token Prediction) với một kho dữ liệu khổng lồ, không cần gán nhãn. Giai đoạn này giúp mô hình học được các “biểu diễn” ngôn ngữ tổng quát, và cũng là phần tốn kém nhất.
    \item \textbf{Tinh chỉnh (fine-tuning).} Lấy mô hình đã được huấn luyện trước và tiếp tục huấn luyện nó trên một bộ dữ liệu nhỏ hơn, đã được gán nhãn, dành riêng cho tác vụ mục tiêu (ví dụ: phân loại cảm xúc).
\end{enumerate}

\begin{center}
    \bookimage[0.9\textwidth]{transfer_learning_paradigm}{Sơ đồ hai giai đoạn: khối bên trái là pre-training, lấy kho văn bản lớn không nhãn làm đầu vào cho một mô hình ngôn ngữ; mũi tên chuyển trọng số của mô hình đó sang khối bên phải, nơi mô hình được tinh chỉnh trên bộ dữ liệu nhỏ có nhãn của một tác vụ cụ thể và cho ra mô hình riêng của tác vụ đó.}
    \captionof{figure}{Học chuyển giao trong NLP: một mô hình được huấn luyện trước trên dữ liệu lớn, sau đó được tinh chỉnh trên một tác vụ cụ thể.}
    \label{fig:transfer_learning_paradigm}
\end{center}

Học chuyển giao có ba lợi ích:
\begin{itemize}
    \item \textbf{Tiết kiệm tài nguyên.} Không phải huấn luyện lại một mô hình khổng lồ từ đầu cho mỗi tác vụ; kiến thức có được từ quá trình pre-training tốn kém được dùng lại.
    \item \textbf{Hiệu năng tốt hơn.} Các biểu diễn ngôn ngữ tổng quát học từ dữ liệu lớn là điểm khởi đầu tốt hơn nhiều so với khởi tạo trọng số ngẫu nhiên, giúp mô hình hội tụ nhanh hơn và đạt độ chính xác cao hơn, nhất là khi dữ liệu cho tác vụ cụ thể có hạn.
    \item \textbf{Hạ rào cản tham gia.} Người phát triển và nhà nghiên cứu không có tài nguyên tính toán khổng lồ vẫn xây dựng được các ứng dụng NLP state-of-the-art bằng cách tinh chỉnh các mô hình đã được huấn luyện trước.
\end{itemize}

\subsection{Tinh chỉnh toàn bộ (Full Fine-tuning)}
\label{ssec:full_finetuning}

Đây là phương pháp tinh chỉnh kinh điển và trực quan nhất.

\subsubsection{Cơ chế hoạt động}
\begin{itemize}
    \item \textbf{Bước 1: Tải mô hình đã huấn luyện trước.} Tải toàn bộ kiến trúc và trọng số của một mô hình như BERT hoặc RoBERTa.
    \item \textbf{Bước 2: Thay thế “đầu” của mô hình.} Mô hình gốc thường có một “đầu” (head) được thiết kế cho tác vụ pre-training (ví dụ, đầu MLM). Chúng ta sẽ loại bỏ cái đầu này và thay thế nó bằng một “đầu” mới, phù hợp với tác vụ hạ nguồn của mình.
        \begin{itemize}
            \item \textbf{Ví dụ (phân loại câu):} Thêm một lớp tuyến tính (linear layer) đơn giản với một lớp softmax lên trên vector biểu diễn của token \texttt{[CLS]}. Số nơ-ron đầu ra của lớp này bằng số lớp cần phân loại.
            \item \textbf{Ví dụ (gán nhãn chuỗi, NER):} Thêm một lớp tuyến tính lên trên vector biểu diễn của \textit{mỗi} token trong chuỗi.
        \end{itemize}
    \item \textbf{Bước 3: Huấn luyện trên dữ liệu của tác vụ mới.} Toàn bộ mô hình (cả phần thân đã được huấn luyện trước và cái đầu mới) được huấn luyện tiếp trên bộ dữ liệu gán nhãn của tác vụ mục tiêu. Tất cả các trọng số, từ lớp embedding dưới cùng đến lớp phân loại trên cùng, đều được cập nhật thông qua thuật toán lan truyền ngược để tối thiểu hóa hàm mất mát trên tác vụ mới.
\end{itemize}
Giai đoạn này thường dùng \textbf{tốc độ học (learning rate) rất nhỏ}, vì các trọng số đã được huấn luyện trước vốn đã ở một điểm khá tốt: mục tiêu là dịch chúng đi một chút cho thích ứng với dữ liệu mới, chứ không phải xoá đi những gì đã học bằng các bước cập nhật quá lớn.

\subsubsection{Chiến lược tinh chỉnh nâng cao}
Ngoài việc chạy lại quá trình huấn luyện, có vài chiến lược giúp tinh chỉnh hiệu quả hơn.
\paragraph{Huấn luyện theo lớp phân biệt (Discriminative Layer-wise Training)}
Các lớp khác nhau của một mô hình học sâu học các loại đặc trưng khác nhau: các lớp dưới cùng (gần input) học các đặc trưng rất chung như cú pháp cơ bản, còn các lớp trên cùng học các đặc trưng trừu tượng và chuyên biệt hơn. Vì vậy, khi tinh chỉnh nên cập nhật các lớp trên cùng nhiều hơn và các lớp dưới cùng ít hơn, bằng cách áp dụng tốc độ học khác nhau cho các lớp khác nhau. Ví dụ, lớp phân loại mới có thể có learning rate là $10^{-3}$, các lớp Transformer trên cùng là $10^{-4}$, và các lớp Transformer dưới cùng là $10^{-5}$.

\paragraph{Rã đông dần dần (Gradual Unfreezing)}
Để tránh “quên thảm khốc” (catastrophic forgetting) -- hiện tượng mô hình nhanh chóng quên đi kiến thức đã học trong quá trình pre-training -- dữ liệu mới được đưa vào từ từ:
\begin{enumerate}
    \item Ban đầu, “đóng băng” (freeze) toàn bộ các lớp đã được huấn luyện trước và chỉ huấn luyện cái đầu phân loại mới trong một vài epoch.
    \item Sau đó, “rã đông” (unfreeze) lớp Transformer trên cùng và huấn luyện cả nó và cái đầu mới.
    \item Tiếp tục như vậy, rã đông dần dần từng lớp từ trên xuống dưới, cho đến khi toàn bộ mô hình được huấn luyện.
\end{enumerate}

\subsubsection{Ưu và nhược điểm của tinh chỉnh toàn bộ}
\begin{assess}{Đánh giá tinh chỉnh toàn bộ (Full Fine-tuning)}
\textbf{Ưu điểm:}
\begin{itemize}
    \item \textbf{Hiệu năng cao nhất:} Thường cho độ chính xác tốt nhất, vì toàn bộ mô hình được tối ưu cho tác vụ mục tiêu.
    \item \textbf{Tương đối đơn giản:} Là một phương pháp đã được thiết lập rõ ràng và dễ triển khai.
\end{itemize}
\textbf{Nhược điểm:}
\begin{itemize}
    \item \textbf{Cực kỳ tốn kém về tài nguyên:} Đây là hạn chế lớn nhất khi làm việc với LLM. Với 10 tác vụ khác nhau, phải lưu 10 bản sao đầy đủ của mô hình đã được tinh chỉnh, mỗi bản sao có thể nặng hàng trăm gigabyte.
    \item \textbf{Dễ bị “quên thảm khốc”:} Nếu dữ liệu tinh chỉnh quá khác biệt so với dữ liệu pre-training, mô hình có thể mất đi các khả năng tổng quát của nó.
    \item \textbf{Rủi ro về bảo mật và ổn định:} Việc cập nhật toàn bộ trọng số có thể sinh ra các hành vi không mong muốn.
\end{itemize}
\end{assess}

\subsection{Tinh chỉnh cho miền chuyên biệt (Domain Adaptation)}
\label{ssec:domain_adaptation}
Một nhu cầu rất phổ biến trong thực tế là đưa mô hình tổng quát vào một \emph{miền} cụ thể: y tế, pháp luật, tài chính, dịch thuật chuyên ngành, hay tiếng Việt. Có hai tầng thích ứng bổ trợ nhau.

\paragraph{1. Tiền huấn luyện tiếp tục (Continued Pretraining).} Gururangan et al. \cite{gururangan2020dontstop} cho thấy tiếp tục huấn luyện mô hình ngôn ngữ trên văn bản \emph{không nhãn} của miền (Domain-Adaptive Pretraining -- DAPT) cải thiện kết quả trên cả bốn miền được thử nghiệm (y sinh, khoa học máy tính, tin tức, đánh giá sản phẩm); tiếp tục huấn luyện trên chính dữ liệu không nhãn của tác vụ (Task-Adaptive Pretraining -- TAPT) mang lại lợi ích bổ sung, và kết hợp cả hai thường tốt nhất. Với LLM hiện đại, khi tiếp tục tiền huấn luyện nên \textbf{trộn lại một phần dữ liệu tổng quát} (replay) để giảm quên thảm khốc, và dùng learning rate nhỏ hơn giai đoạn tiền huấn luyện gốc.

\paragraph{2. Tinh chỉnh có giám sát trên dữ liệu miền.} Sau đó (hoặc thay vì bước trên, nếu dữ liệu ít), tinh chỉnh trên các cặp chỉ dẫn--đáp án của miền. Moslem et al. \cite{moslem2023adaptivemt} tinh chỉnh Mistral 7B trên chỉ 20\,000 đoạn song ngữ Tây Ban Nha--Anh thuộc miền y tế, trộn lẫn prompt zero-shot và one-shot (one-shot nghĩa là prompt có kèm một cặp câu dịch mẫu tương tự). Kết quả:
\begin{itemize}
	\item Mô hình tinh chỉnh vượt \texttt{gpt-3.5-turbo} ở chế độ zero-shot và ngang ở chế độ one-shot;
	\item Ngang NLLB 3.3B (mô hình dịch chuyên dụng) ở zero-shot và vượt nó ở one-shot;
	\item Tinh chỉnh với định dạng one-shot giúp mô hình \emph{biết tận dụng ví dụ trong ngữ cảnh} -- điều kiện cho dịch thích ứng thời gian thực, nơi các câu dịch tương tự được truy xuất từ bộ nhớ dịch rồi đưa vào prompt.
\end{itemize}

\begin{keyidea}{Khi nào tinh chỉnh, khi nào dùng RAG hay prompt?}
\begin{itemize}
	\item \textbf{Cần kiến thức cập nhật, có nguồn trích dẫn} $\rightarrow$ RAG (chương~\ref{chap:rag}).
	\item \textbf{Cần định dạng, văn phong, hành vi ổn định; hoặc cần mô hình nhỏ rẻ thay cho mô hình lớn} $\rightarrow$ tinh chỉnh (thường bằng PEFT).
	\item \textbf{Miền có ngôn ngữ/thuật ngữ rất khác văn bản web} $\rightarrow$ cân nhắc tiền huấn luyện tiếp tục trước khi tinh chỉnh.
	\item \textbf{Chưa có dữ liệu, cần thử nhanh} $\rightarrow$ bắt đầu bằng prompt engineering và dùng nó làm baseline.
\end{itemize}
\end{keyidea}

Dù là tinh chỉnh tổng quát hay thích ứng miền, chi phí lưu trữ và tính toán của tinh chỉnh toàn bộ trở nên khó chấp nhận ở quy mô hàng tỷ tham số. Đó là động lực của hướng \textbf{tinh chỉnh hiệu quả tham số (Parameter-Efficient Fine-Tuning -- PEFT)}, nội dung của mục tiếp theo.
