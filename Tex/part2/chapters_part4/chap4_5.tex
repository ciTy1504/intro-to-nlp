\section{Recipe 5: xây dựng một Agent đơn giản có khả năng sử dụng công cụ}
\label{sec:recipe_agent}

\textbf{Mục tiêu:} Xây dựng một tác tử (Agent) trả lời được các câu hỏi về sự kiện gần đây hoặc chủ đề chuyên sâu, bằng cách tự quyết định khi nào cần tìm thông tin trên Wikipedia.

\textbf{Thành phần chính:}
\begin{itemize}
    \item \textbf{\texttt{LangChain}:} Cung cấp các công cụ để xây dựng các agent.
    \item \textbf{LLM:} Thành phần ra quyết định của agent.
    \item \textbf{Công cụ (Tool):} Một API hoặc thư viện để tương tác với thế giới bên ngoài (trong trường hợp này là Wikipedia).
\end{itemize}

\textbf{Các bước thực hiện:}
\begin{enumerate}
    \item \textbf{Thiết lập môi trường:} Cài đặt thư viện và API keys.
    \item \textbf{Định nghĩa công cụ:} Tạo một hoặc nhiều công cụ mà agent có thể sử dụng.
    \item \textbf{Khởi tạo LLM và agent:} Chọn một LLM và dùng các hàm của LangChain để gắn công cụ cho nó.
    \item \textbf{Thực thi agent:} Đưa ra một câu hỏi, rồi đọc lại chuỗi tin nhắn mà agent trả về để thấy vòng lặp \texttt{Thought-\allowbreak Action-\allowbreak Observation}.
\end{enumerate}

\textbf{Mã nguồn:}
\begin{example}{Agent tra cứu Wikipedia}{ex:simple_agent}
    \begin{minted}{python}
    # Bước 1: Cài đặt
    # pip install -U langchain wikipedia
    # và gói của nhà cung cấp mô hình, ví dụ: pip install -U langchain-openai
    import os
    # Khóa API đọc từ biến môi trường của nhà cung cấp, ví dụ
    # OPENAI_API_KEY hoặc ANTHROPIC_API_KEY
    # os.environ["OPENAI_API_KEY"] = "..."

    from langchain.agents import create_agent
    from langchain.chat_models import init_chat_model
    from langchain.messages import AIMessage, ToolMessage
    from langchain.tools import tool

    # Bước 2: Định nghĩa Công cụ
    # Tên công cụ lấy từ tên hàm, phần mô tả lấy từ docstring; mô hình chỉ
    # dựa vào hai thứ đó để quyết định khi nào gọi.
    @tool
    def search_wikipedia(query: str) -> str:
        """Searches Wikipedia for a short summary of the given query.
        Use this for questions about entities, facts, or specific events.
        """
        from wikipedia import summary
        try:
            return summary(query, sentences=2)
        except Exception as e:
            return f"Error searching Wikipedia: {e}"

    tools = [search_wikipedia]

    # Bước 3: Khởi tạo LLM và Agent
    # Điền model id hiện hành của nhà cung cấp (xem tài liệu của họ trước khi chạy)
    MODEL = "<model-id>"
    llm = init_chat_model(MODEL, model_provider="<provider>")

    # create_agent trả về một graph đã biên dịch: nó tự lặp giữa lượt gọi
    # mô hình và lượt chạy công cụ cho tới khi mô hình không gọi thêm.
    agent = create_agent(
        model=llm,
        tools=tools,
        system_prompt=(
            "You are a research assistant. If a question needs facts you "
            "are not sure about, call search_wikipedia before answering. "
            "Answer in two or three sentences."
        ),
    )

    # Bước 4: Thực thi Agent
    query = "What is the main idea behind the LLaVA model?"
    result = agent.invoke({"messages": [{"role": "user", "content": query}]})

    # Cả vòng lặp nằm trong result["messages"]: mỗi lần mô hình quyết định
    # dùng công cụ là một AIMessage có tool_calls, mỗi kết quả trả về là
    # một ToolMessage.
    for msg in result["messages"]:
        if isinstance(msg, AIMessage) and msg.tool_calls:
            for call in msg.tool_calls:
                print(f"[Action] {call['name']} {call['args']}")
        elif isinstance(msg, ToolMessage):
            print(f"[Observation] {msg.content}")

    print("\nFinal Answer:")
    print(result["messages"][-1].text)
    \end{minted}
\end{example}
\textbf{Kết quả mong đợi:} Phần in ra ở bước 4 cho thấy agent đã gọi công cụ, nhận kết quả, rồi mới trả lời:
\begin{verbatim}
[Action] search_wikipedia {'query': 'LLaVA model'}
[Observation] LLaVA (Large Language and Vision Assistant) is a large
multimodal model (LMM) created by researchers from the University of
Wisconsin-Madison, Microsoft Research, and Columbia University. It
combines a vision encoder with a large language model (LLM) to enable
multimodal chat capabilities.

Final Answer:
The main idea behind the LLaVA model is to combine a vision encoder
with a large language model, so that one model can both read an image
and discuss it in a conversation.
\end{verbatim}
Bước \texttt{Thought} không còn được in ra dưới dạng văn bản: mô hình phát ra một lời gọi công cụ có cấu trúc, nên phần suy luận nằm trong quyết định gọi \texttt{search\_wikipedia} chứ không nằm trong chuỗi ký tự mà mô hình sinh ra.

\begin{deepdive}{Cập nhật 2026: tác tử hiện đại không còn mô phỏng ReAct bằng văn bản tự do}
Công thức ReAct ở trên có ích để \emph{hiểu} vòng lặp Suy nghĩ--Hành động--Quan sát (mục~\ref{ssec:react_framework}), nhưng cách triển khai thực tế hiện nay khác ở ba điểm:
\begin{enumerate}
	\item \textbf{Gọi công cụ có cấu trúc (native tool calling):} Thay vì bắt mô hình in ra chuỗi \texttt{Action: ...} rồi tự phân tích bằng regex (rất dễ vỡ), ta khai báo công cụ dưới dạng \emph{JSON schema} và mô hình trả về lời gọi công cụ có cấu trúc. Mọi API hiện đại và các engine mở (vLLM, SGLang) đều hỗ trợ, thường với ràng buộc ngữ pháp để đảm bảo JSON hợp lệ.
	\item \textbf{Model Context Protocol (MCP):} Một chuẩn mở để mô tả và kết nối công cụ/nguồn dữ liệu với LLM. Thay vì viết lại adapter cho từng framework, bạn dựng một “MCP server” cho hệ thống của mình và mọi client hỗ trợ MCP đều dùng được.
	\item \textbf{Mô hình suy luận thay cho prompt ReAct:} Các mô hình suy luận (chương~\ref{chap:reasoning}) đã được huấn luyện để tự lập kế hoạch và tự sửa sai; phần “Thought” viết tay trong prompt trở nên ít cần thiết. Việc cần làm nhiều hơn là thiết kế \textbf{bộ công cụ tốt}, \textbf{giới hạn số vòng lặp}, và \textbf{cơ chế phê duyệt} cho các hành động khó đảo ngược.
\end{enumerate}
Khi đánh giá tác tử, hãy dùng các benchmark chuyên biệt (mục~\ref{sec:agent_evaluation}) và nhớ rằng điểm số phụ thuộc mạnh vào “giàn giáo” quanh mô hình, không chỉ vào mô hình.
\end{deepdive}
