			\fcolorbox{orange!70!black}{orange!5}{\parbox{\dimexpr#1-2\fboxsep-2\fboxrule\relax}{\centering\small
					\textbf{[Hình đang chờ bổ sung: \texttt{\detokenize{#2}}]}\\[2pt]\textit{#3}}}}}}
