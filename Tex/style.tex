% 7. QUY ƯỚC TRÌNH BÀY CỦA SÁCH  (dùng đúng 5 loại hộp, không tự chế màu mới)
% ==============================================================================
%
%   \begin{keyidea}{Tiêu đề}   ... \end{keyidea}
%       Trực giác cốt lõi, ý tưởng then chốt, kết luận quan trọng của một mục.
%
%   \begin{pitfall}{Tiêu đề}   ... \end{pitfall}
%       Cạm bẫy, cảnh báo, nguyên tắc không được vi phạm, lỗi hay gặp.
%
%   \begin{deepdive}{Tiêu đề}  ... \end{deepdive}
%       Đọc thêm, ghi chú đi sâu, dẫn giải toán học có thể bỏ qua ở lần đọc đầu.
%
%   \begin{assess}{Tiêu đề}    ... \end{assess}
%       Đánh giá / so sánh một phương pháp, tổng kết một nhóm kỹ thuật.
%
%   \begin{checklist}{Tiêu đề} ... \end{checklist}
%       Danh sách kiểm, quy trình thao tác theo bước.
%
% HÌNH:  luôn dùng khuôn
%       \begin{center}
%       \bookimage[0.9\textwidth]{ten-file}{mô tả nội dung hình cho người không xem được}
%       \captionof{figure}{Chú thích đầy đủ, nói ý nghĩa chứ không mô tả lại hình.}
%       \label{fig:...}
%       \end{center}
%
% BẢNG:  bảng được nhắc tới trong văn bản thì phải có số và nhãn:
%       \begin{table}[H] \centering \small ... \caption{...} \label{tab:...} \end{table}
%       Bảng nhỏ chỉ để minh họa tại chỗ thì dùng \begin{center}...\end{center}, không đánh số.
%
% MÃ VÀ CHUỖI NGUYÊN VĂN:
%       \texttt{...}  cho mọi thứ in ra nguyên văn: tên hàm, tên gói, tên cờ dòng lệnh,
%                     token và mảnh subword (<SOS>, [CLS], </w>), chuỗi đầu vào/đầu ra
%                     của mô hình, prompt mẫu ngắn.
%       \textit{...}  cho một từ của ngôn ngữ tự nhiên được dẫn ra làm ví dụ ngôn ngữ học
%                     (bàn về nghĩa hay hình thái của chính từ đó).
%       KHÔNG dùng dấu nháy ngược kiểu markdown: trong LaTeX `x` in ra thành ‘x,
%       không ra font mã. Ngoặc nhọn trong \texttt phải thoát: \texttt{\{a, b\}},
%       nếu không LaTeX coi là nhóm và cặp ngoặc sẽ không được in ra.
%       Trích dẫn trong văn bản dùng ``...'' (hai nháy ngược + hai nháy đơn).
% ==============================================================================
